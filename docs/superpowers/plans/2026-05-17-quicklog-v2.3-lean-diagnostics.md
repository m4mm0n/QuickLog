# QuickLog v2.3 Lean Diagnostics Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add every approved lean diagnostic idea while keeping QuickLog dependency-free, engine-friendly, and smaller in spirit than a general logging framework.

**Architecture:** Keep the core library focused on small primitives: configuration presets, validation, session markers, rate limiting, crash fingerprints, runtime state snapshots, and compact formatting controls. Put analysis-heavy behavior in `QuickLog.Tools` as offline commands that reuse existing readers/exporters and write static files only.

**Tech Stack:** C#/.NET SDK only, existing xUnit test project, BCL-only APIs (`System.IO`, `System.Text.Json`, `System.Diagnostics`, `System.IO.Compression`, `System.Collections.Concurrent`, `System.Security.Cryptography` if a built-in hash is needed).

---

## Non-Negotiable Guardrails

- Do not add any runtime `PackageReference` to `QuickLog`, `QuickLog.Tools`, or `QuickLog.Sample`.
- Do not add DI, message templates, OpenTelemetry, Serilog-style abstractions, hosting integrations, or live web dashboards to core.
- Keep new public APIs additive and optional.
- Keep core helpers tiny and synchronous unless async is already required by the existing pipeline.
- Keep report/dashboard output as static HTML generated by `QuickLog.Tools`; no server and no JavaScript framework.
- Prefer one small type per feature cluster over expanding `QuickLogger` into a large multi-purpose object.
- After each phase, run `dotnet test QuickLog.sln -c Release`.
- After each phase, run `rg -n "<PackageReference" QuickLog QuickLog.Tools QuickLog.Sample` and verify there are no runtime package references.

```csharp
namespace QuickLog.Core;

public sealed class LogSession : IDisposable
{
    private readonly IQuickLog _logger;
    private readonly string _name;
    private readonly string _sessionId;
    private readonly DateTime _startedUtc;
    private bool _disposed;

    private LogSession(IQuickLog logger, string name, string sessionId)
    {
        _logger = logger;
        _name = name;
        _sessionId = sessionId;
        _startedUtc = DateTime.UtcNow;
        _logger.Log(LogType.Info, $"SESSION BEGIN {_name} session={_sessionId}");
    }

    public static LogSession Begin(IQuickLog logger, string name, string? sessionId = null)
        => new(logger, name, sessionId ?? Guid.NewGuid().ToString("N"));

    public void Checkpoint(string name)
        => _logger.Log(LogType.Info, $"CHECKPOINT {name} session={_sessionId}");

    public void Bookmark(string name)
        => _logger.Log(LogType.Info, $"BOOKMARK {name} session={_sessionId}");

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;
        var ms = (long)(DateTime.UtcNow - _startedUtc).TotalMilliseconds;
        _logger.Log(LogType.Info, $"SESSION END {_name} session={_sessionId} durationMs={ms}");
    }
}
```

```csharp
using System.Collections.Concurrent;

namespace QuickLog.Core;

internal sealed class LogRateLimiter
{
    private readonly ConcurrentDictionary<string, byte> _once = new(StringComparer.Ordinal);
    private readonly ConcurrentDictionary<string, DateTime> _last = new(StringComparer.Ordinal);

    public bool ShouldLogOnce(string key) => _once.TryAdd(key, 0);

    public bool ShouldLogEvery(string key, TimeSpan interval)
    {
        var now = DateTime.UtcNow;
        return _last.AddOrUpdate(key, now, (_, previous) =>
        {
            if (now - previous >= interval)
                return now;
            return previous;
        }) == now;
    }
}
```

```csharp
using System.Security.Cryptography;
using System.Text;

namespace QuickLog.Core;

public static class CrashFingerprint
{
    public static string From(Exception exception)
    {
        var stack = new System.Diagnostics.StackTrace(exception, false);
        var frames = stack.GetFrames()?.Take(6)
            .Select(f => f.GetMethod())
            .Where(m => m is not null)
            .Select(m => $"{m!.DeclaringType?.FullName}.{m.Name}") ?? [];

        var basis = $"{exception.GetType().FullName}|{string.Join("|", frames)}";
        var hash = SHA256.HashData(Encoding.UTF8.GetBytes(basis));
        return Convert.ToHexString(hash.AsSpan(0, 8));
    }
}
```

```csharp
namespace QuickLog.Utilities;

public static class SafeLogPath
{
    public static string SafeFileName(string name)
    {
        var invalid = Path.GetInvalidFileNameChars();
        var cleaned = new string(name.Select(ch => invalid.Contains(ch) ? '_' : ch).ToArray()).Trim();
        return string.IsNullOrWhiteSpace(cleaned) ? "quicklog" : cleaned;
    }

    public static string CreateSessionDirectory(string root, string sessionName)
    {
        var stamp = DateTime.UtcNow.ToString("yyyyMMdd-HHmmss");
        var path = Path.Combine(root, $"{stamp}-{SafeFileName(sessionName)}");
        Directory.CreateDirectory(path);
        return path;
    }
}
```
